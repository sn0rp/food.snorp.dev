\smallheading{Ingredients}
\textbf{Noodles \& sauce}
\begin{itemize}
\item 250 g Korean sweet potato glass noodles
\item 2 tbsp soy sauce, 1 tbsp sesame oil, \(\frac12\) tbsp sugar (noodle sauce)
\item 2 tbsp soy sauce, 1 tbsp sesame oil, \(\frac12\) tbsp sugar, 1 tsp minced garlic, black pepper to taste (finishing sauce)
\end{itemize}
\textbf{Protein \& marinade}
\begin{itemize}
\item 1lb 85/15 beef ground beef
\item 3 tbsp soy sauce, 1 tbsp honey, 2 tsp garlic, 2 tbsp sesame oil, black pepper to taste
\end{itemize}
\textbf{Vegetables \& garnish}
\begin{itemize}
\item 1 onion, 2 carrots, 1 bell pepper, sliced shiitake mushrooms to taste
\item Toasted sesame seeds, green onion, toasted sesame oil for finishing
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.35\linewidth}
\smallheading{Tools \& alternatives}
\begin{itemize}
\item Large pot \textit{(for noodles)}
\item Wok or large skillet \textit{(nonstick skillet works)}
\item Mixing bowls \textit{(or reusable containers)}
\item Colander or strainer
\item Knife and cutting board \textit{(mandoline or peeler optional)}
\item Tongs or chopsticks
\end{itemize}
\vspace{5pt}
{\footnotesize\color{muted}\textbf{MAKE-AHEAD NOTE}\par The original recipe soaks noodles for 2 hours}
\end{minipage}
%\vspace{2pt}
\smallheading{Directions}
\begin{enumerate}
\item \textbf{Set up.} Mix the noodle sauce, finishing sauce, and marinade in separate bowls. Slice vegetables into thin, similar-sized strips. Fill a large pot with water.
\item \textbf{Marinate.} Mix beef with marinade and set aside while preparing the remaining ingredients.
\item \textbf{Cook noodles.} Bring water to a boil; cook soaked noodles for about 5 minutes (or per package). Drain without rinsing. Cut shorter if desired. Toss in a hot pan with noodle sauce for 1--2 minutes; transfer to a large bowl.
\item \textbf{Cook beef and vegetables.} Stir-fry protein until safely cooked through; set aside. Cook vegetables separately or in batches until just tender.
\item \textbf{Combine and serve.} Toss noodles, beef, and vegetables with finishing sauce. Finish with sesame seeds, green onion, and a little sesame oil. Chili crunch works well here too.
\end{enumerate}
