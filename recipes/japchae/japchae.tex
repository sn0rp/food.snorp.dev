\smallheading{Ingredients}
\textbf{Noodles \& sauce}
\begin{itemize}
\item 250g Korean sweet potato glass noodles
\item 2tbsp soy sauce, 1tbsp sesame oil, \(\frac12\)tbsp sugar (noodle sauce)
\item 2tbsp soy sauce, 1tbsp sesame oil, \(\frac12\)tbsp sugar, 1tsp garlic, black pepper to taste (finishing sauce)
\end{itemize}
\textbf{Protein \& marinade}
\begin{itemize}
\item 1lb 85/15 ground beef
\item 3 tbsp soy sauce, 1 tbsp honey, 2 tsp garlic, 2 tbsp sesame oil, black pepper to taste
\end{itemize}
\textbf{Vegetables}
\begin{itemize}
\item 1 onion
\item 2 carrots
\item 1 bell pepper
\item 5oz sliced shiitake mushrooms
\item Neutral oil
\end{itemize}
\textbf{Garnish}
\begin{itemize}
\item Toasted sesame seeds
\item Green onion
\item Toasted sesame oil or chili crunch
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.35\linewidth}

\item Wok or large skillet

\smallheading{Directions}
\begin{enumerate}
\item \textbf{Prepare ingredients.} Mix the noodle sauce, finishing sauce, and marinade in separate bowls. Fill a large pot with water. Dice the onion, slice the bell peppers and shiitake mushrooms into bite-size strips, peel the carrots into ribbons, then cut the ribbons into shorter lengths and slice the green onion. Keep the vegetables separate.
\item \textbf{Cook noodles.} Bring water to a boil; cook noodles for about 5 minutes (or per package). Drain and set aside.
\item \textbf{Cook beef.} Cook the beef. Heat a large skillet or wok over medium-high heat. Add the ground beef and marinade, and cook until the beef is fully cooked, breaking it up as it cooks. Transfer it to a large bowl.
\item \textbf{Cook vegetables.} Stir-fry each vegetable separately; add oil as needed. Transfer to the bowl with the beef.
\item \textbf{Season noodles.} Add the drained noodles and noodle sauce to the hot skillet or wok. Toss for 1–2 minutes, until the noodles are heated through and coated with sauce.
\item \textbf{Combine.} Return the beef and vegetables to the skillet along with finishing sauce and toss until everything is heated through and evenly coated.
\item \textbf{Serve.} Garnish with green onion, sesame seeds, and sesame oil or chili crunch.
\end{enumerate}
